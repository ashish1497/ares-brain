# A-ingest Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Normalize the scraped `raw/` tree plus user-dropped `inbox/` files into `courses/<slug>/normalized/*.md` (the corpus sub-project B will embed), and transcribe class recordings into `courses/<slug>/transcripts/*.md`.

**Architecture:** Same shape as A-scrape — pure-Python functions in `scraper/`, driven by new `lms_scrape.py` subcommands (`transcribe`, `ingest`) that also join the `all` orchestrator, exposed through two new deterministic MCP tools and a `/course-ingest` command. No LLM anywhere.

**Tech Stack:** Python 3.12 (`uv` venv) + `pypdf`, `python-docx`, `python-pptx`, `openpyxl`, `markdownify`, `yt-dlp`, `faster-whisper`; existing TypeScript MCP server (`@modelcontextprotocol/sdk@1.30.0`); `pytest` / `vitest`.

- `COURSE_AGENT_HOME` overrides the data-tree root; always resolve paths through `paths.py` (`HOME`, `course_dir`, `raw_dir`, `global_file`, `ensure`, `slugify`). `paths.HOME` is module-level and re-derived on `importlib.reload` (the `home` test fixture in `scraper/tests/conftest.py` relies on this).
- Every subcommand/step is idempotent and resumable; one failure never aborts the run. Follow the `_run_all` per-item try/except pattern already in `scraper/lms_scrape.py`.
- All JSON writes go through `scrape_steps._write_json(path, data)` (creates parent dirs, `json.dumps(indent=1)`).
- Transcription model: `faster-whisper`, `WhisperModel(os.environ.get("WHISPER_MODEL", "small"), device="auto", compute_type="int8")`.
- HTML → markdown: `markdownify`, links preserved.
- Books: PDF/DOCX only. `.epub` is skipped with a logged message — no `ebooklib` dependency.
- Normalized-file frontmatter keys (exact): `type, source, title, course, session?, due?, recordedOn?, bodyHash, updatedAt`. `type` ∈ `material | assignment | outline | transcript | self-note | book | session | announcement`.
- `bodyHash` = `hashlib.sha256(body.encode()).hexdigest()` (hash of the rendered markdown body).
- Manifest (`normalized/_ingest.json`) `hash` = sha256 of the source's **input** (file bytes; `json.dumps(entry, sort_keys=True)` for API entries; fetched CSV text for the outline) — distinct from `bodyHash`.

```python
"""HTML content fields (assignment instructions, material bodies, announcements)
to markdown. Links are kept — the Google Sheets / workbook URLs matter."""
import re

from markdownify import markdownify


def html_to_markdown(html: str) -> str:
    if not html or not html.strip():
        return ""
    md = markdownify(html, heading_style="ATX", strip=["span"])
    md = re.sub(r"\n{3,}", "\n\n", md)
    return md.strip()
```

```python
"""Extract plain text from course files: PDF, DOCX, PPTX, XLSX."""
from pathlib import Path


class UnsupportedFormat(Exception):
    pass


def _pdf(path: Path) -> str:
    from pypdf import PdfReader
    reader = PdfReader(str(path))
    return "\n\n".join((page.extract_text() or "") for page in reader.pages).strip()


def _docx(path: Path) -> str:
    from docx import Document
    doc = Document(str(path))
    parts = [p.text for p in doc.paragraphs if p.text.strip()]
    for table in doc.tables:
        for row in table.rows:
            parts.append("\t".join(c.text for c in row.cells))
    return "\n".join(parts).strip()


def _pptx(path: Path) -> str:
    from pptx import Presentation
    prs = Presentation(str(path))
    out = []
    for i, slide in enumerate(prs.slides, 1):
        out.append(f"--- Slide {i} ---")
        for shape in slide.shapes:
            if shape.has_text_frame and shape.text_frame.text.strip():
                out.append(shape.text_frame.text.strip())
        if slide.has_notes_slide and slide.notes_slide.notes_text_frame.text.strip():
            out.append("Notes: " + slide.notes_slide.notes_text_frame.text.strip())
    return "\n".join(out).strip()


def _xlsx(path: Path) -> str:
    from openpyxl import load_workbook
    wb = load_workbook(str(path), read_only=True, data_only=True)
    out = []
    for ws in wb.worksheets:
        out.append(f"--- Sheet: {ws.title} ---")
        for row in ws.iter_rows(values_only=True):
            cells = ["" if v is None else str(v) for v in row]
            if any(cells):
                out.append("\t".join(cells))
    wb.close()
    return "\n".join(out).strip()


_READERS = {".pdf": _pdf, ".docx": _docx, ".pptx": _pptx, ".xlsx": _xlsx}


def extract_text(path: Path) -> str:
    reader = _READERS.get(path.suffix.lower())
    if reader is None:
        raise UnsupportedFormat(f"no extractor for {path.suffix!r} ({path.name})")
    return reader(path)
```

```python
"""Fetch a link-shared Google Sheet (the course outline) via its CSV export.

The LMS stores the outline as an HTML `text` material whose only content is an
<a href> to a Google Sheet. Link-shared sheets export without auth at
https://docs.google.com/spreadsheets/d/<id>/export?format=csv .
"""
import csv
import io
import re

import requests

_ID_RE = re.compile(r"docs\.google\.com/spreadsheets/d/([a-zA-Z0-9_-]+)")
_TIMEOUT = 30


def sheet_id_from_html(html: str) -> str | None:
    m = _ID_RE.search(html or "")
    return m.group(1) if m else None


def fetch_csv(sheet_id: str) -> str | None:
    url = f"https://docs.google.com/spreadsheets/d/{sheet_id}/export?format=csv"
    try:
        resp = requests.get(url, timeout=_TIMEOUT)
    except requests.RequestException:
        return None
    if resp.status_code != 200 or not resp.text.strip():
        return None
    return resp.text


def csv_to_markdown_table(csv_text: str) -> str:
    rows = list(csv.reader(io.StringIO(csv_text)))
    rows = [r for r in rows if any(cell.strip() for cell in r)]
    if not rows:
        return ""
    width = max(len(r) for r in rows)
    rows = [r + [""] * (width - len(r)) for r in rows]
    header, *body = rows
    out = ["| " + " | ".join(header) + " |",
           "| " + " | ".join("---" for _ in header) + " |"]
    for r in body:
        out.append("| " + " | ".join(cell.replace("|", "\\|") for cell in r) + " |")
    return "\n".join(out)
```

```python
"""Transcribe class recordings: pull audio from the YouTube link with yt-dlp,
run faster-whisper. All recordings are YouTube links (see docs/lms-api.md)."""
import json
import os
import subprocess
import tempfile
from pathlib import Path

from paths import course_dir, raw_dir, ensure
from scrape_steps import _write_json, _read_or_none

_MODEL = None


def _fmt_ts(seconds: float) -> str:
    s = int(seconds)
    return f"{s // 3600}:{(s % 3600) // 60:02d}:{s % 60:02d}"


def _pull_audio(url: str, dest_dir: Path) -> Path | None:
    out_tmpl = str(dest_dir / "%(id)s.%(ext)s")
    try:
        subprocess.run(
            ["yt-dlp", "-x", "--audio-format", "m4a", "--no-playlist",
             "-o", out_tmpl, url],
            check=True, capture_output=True, timeout=900,
        )
    except (subprocess.CalledProcessError, subprocess.TimeoutExpired, FileNotFoundError):
        return None
    files = list(dest_dir.glob("*.m4a"))
    return files[0] if files else None


def _get_model():
    global _MODEL
    if _MODEL is None:
        from faster_whisper import WhisperModel
        _MODEL = WhisperModel(
            os.environ.get("WHISPER_MODEL", "small"),
            device="auto", compute_type="int8",
        )
    return _MODEL


def _whisper(audio: Path) -> list[tuple[float, str]]:
    segments, _info = _get_model().transcribe(str(audio), vad_filter=True)
    return [(seg.start, seg.text.strip()) for seg in segments if seg.text.strip()]


def _write_transcript(dest: Path, rec: dict, course: dict, segs: list[tuple[float, str]]) -> None:
    body = "\n".join(f"[{_fmt_ts(start)}] {text}" for start, text in segs)
    fm = [
        "---",
        "type: transcript",
        f"source: raw/recordings.json#{rec['id']}",
        f"title: {rec.get('title', rec['id'])}",
        f"course: {course['name']}",
    ]
    if rec.get("recordedOn"):
        fm.append(f"recordedOn: {rec['recordedOn']}")
    fm += ["---", ""]
    ensure(dest.parent)
    dest.write_text("\n".join(fm) + body + "\n")


def step_transcribe(index: list[dict], course: str | None = None) -> dict:
    scope = [c for c in index if course is None or c["slug"] == course]
    transcribed, skipped, failed = [], [], []
    for c in scope:
        tdir = course_dir(c["slug"]) / "transcripts"
        recs = _read_or_none(raw_dir(c["slug"]) / "recordings.json") or []
        inbox_recs = sorted((course_dir(c["slug"]) / "inbox" / "recordings").glob("*")) \
            if (course_dir(c["slug"]) / "inbox" / "recordings").exists() else []
        failed_here = []
        for rec in recs:
            dest = tdir / f"{rec['id']}.md"
            if dest.exists():
                skipped.append(rec["id"]); continue
            with tempfile.TemporaryDirectory() as tmp:
                audio = _pull_audio(rec.get("videoUrl", ""), Path(tmp))
                if audio is None:
                    failed.append({"id": rec["id"], "status": "needs-manual"})
                    failed_here.append({"id": rec["id"], "status": "needs-manual",
                                        "title": rec.get("title")})
                    continue
                segs = _whisper(audio)
            _write_transcript(dest, rec, c, segs)
            transcribed.append(rec["id"])
        for f in inbox_recs:
            dest = tdir / f"inbox-{f.stem}.md"
            if dest.exists():
                skipped.append(f"inbox-{f.stem}"); continue
            segs = _whisper(f)
            _write_transcript(dest, {"id": f"inbox-{f.stem}", "title": f.name}, c, segs)
            transcribed.append(f"inbox-{f.stem}")
        if failed_here:
            ensure(tdir)
            _write_json(tdir / "_failed.json", failed_here)
    return {"transcribed": transcribed, "skipped": skipped, "failed": failed}
```

```python
"""Normalize the scraped raw/ tree + dropped inbox/ files into
courses/<slug>/normalized/*.md — the corpus sub-project B embeds."""
import hashlib
import json
from datetime import datetime, timezone
from pathlib import Path

from paths import course_dir, courses_root, ensure, slugify
from scrape_steps import _read_or_none


def _now() -> str:
    return datetime.now(timezone.utc).strftime("%Y-%m-%dT%H:%M:%SZ")


def _input_hash(data) -> str:
    if isinstance(data, str):
        data = data.encode()
    return hashlib.sha256(data).hexdigest()


def _body_hash(body: str) -> str:
    return hashlib.sha256(body.encode()).hexdigest()


def _norm_dir(slug: str) -> Path:
    return course_dir(slug) / "normalized"


def _load_manifest(slug: str) -> dict:
    m = _read_or_none(_norm_dir(slug) / "_ingest.json")
    return m if m else {"sources": {}, "ingestedAt": None}


def _save_manifest(slug: str, manifest: dict) -> None:
    ensure(_norm_dir(slug))
    (_norm_dir(slug) / "_ingest.json").write_text(json.dumps(manifest, indent=1))


# in-process record of which stableName maps to which source, per run, per slug
_name_owner: dict[tuple[str, str], str] = {}


def write_normalized(slug: str, *, type_: str, source: str, title: str,
                     course_name: str, body: str, session: int | None = None,
                     due: str | None = None, recorded_on: str | None = None) -> Path:
    out = ensure(_norm_dir(slug))
    base = f"{type_}-{slugify(title or source)}"
    name = f"{base}.md"
    n = 2
    while (slug, name) in _name_owner and _name_owner[(slug, name)] != source:
        name = f"{base}-{n}.md"
        n += 1
    _name_owner[(slug, name)] = source

    fm = ["---", f"type: {type_}", f"source: {source}", f"title: {title}",
          f"course: {course_name}"]
    if session is not None:
        fm.append(f"session: {session}")
    if due:
        fm.append(f"due: {due}")
    if recorded_on:
        fm.append(f"recordedOn: {recorded_on}")
    fm.append(f"bodyHash: {_body_hash(body)}")
    fm.append(f"updatedAt: {_now()}")
    fm += ["---", ""]
    dest = out / name
    dest.write_text("\n".join(fm) + body.rstrip() + "\n")
    return dest


def ensure_inbox_skeleton(index: list[dict]) -> None:
    for c in index:
        for sub in ("notes", "recordings", "books"):
            ensure(course_dir(c["slug"]) / "inbox" / sub)
    ensure(courses_root() / "inbox" / "_unsorted")
```

```python
from extract import extract_text, UnsupportedFormat
from html2md import html_to_markdown
from paths import raw_dir


def _material_body(m: dict, slug: str) -> str:
    kind = m.get("kind")
    if kind == "file" and m.get("localPath"):
        try:
            return extract_text(course_dir(slug) / m["localPath"])
        except UnsupportedFormat as exc:
            return f"(unextractable file: {exc})"
    if kind == "text":
        return html_to_markdown(m.get("content") or "")
    if kind == "link":
        return f"> {m.get('url') or '(no url)'}"
    return ""


def _normalize_materials(slug: str, course_name: str):
    mats = _read_or_none(raw_dir(slug) / "materials.json") or []
    out = []
    for m in mats:
        key = f"api:material:{m['id']}"
        body = _material_body(m, slug)
        h = _input_hash(json.dumps(m, sort_keys=True))
        path = write_normalized(
            slug, type_="material", source=key, title=m.get("title") or m["id"],
            course_name=course_name, body=body,
            session=m.get("session") if isinstance(m.get("session"), int) else None)
        out.append((key, h, path))
    return out


def _normalize_assignments(slug: str, course_name: str):
    items = _read_or_none(raw_dir(slug) / "assignments.json") or []
    out = []
    for a in items:
        key = f"api:assignment:{a['id']}"
        body = html_to_markdown(a.get("instructions") or a.get("description") or "")
        facts = []
        if a.get("dueAt"):
            facts.append(f"Due: {a['dueAt']}")
        if a.get("cutoffDate"):
            facts.append(f"Cutoff: {a['cutoffDate']}")
        if a.get("submissionType"):
            facts.append(f"Submission: {a['submissionType']}")
        facts.append(f"Group: {'yes' if a.get('isGroup') else 'no'}")
        full = (body + "\n\n" + "\n".join(facts)).strip()
        h = _input_hash(json.dumps(a, sort_keys=True))
        path = write_normalized(
            slug, type_="assignment", source=key,
            title=a.get("title") or a["id"], course_name=course_name,
            body=full, due=a.get("dueAt"))
        out.append((key, h, path))
    return out


def _normalize_transcripts(slug: str, course_name: str):
    tdir = course_dir(slug) / "transcripts"
    out = []
    if not tdir.exists():
        return out
    for f in sorted(tdir.glob("*.md")):
        if f.name.startswith("_"):
            continue
        key = f"transcript:{f.stem}"
        raw_text = f.read_text()
        body = raw_text.split("---\n", 2)[-1].strip() if raw_text.startswith("---") else raw_text
        h = _input_hash(raw_text)
        path = write_normalized(
            slug, type_="transcript", source=f"transcripts/{f.name}",
            title=f.stem, course_name=course_name, body=body)
        out.append((key, h, path))
    return out


_NORMALIZERS_API = (_normalize_materials, _normalize_assignments, _normalize_transcripts)


def step_ingest(index: list[dict], course: str | None = None) -> dict:
    _name_owner.clear()
    scope = [c for c in index if course is None or c["slug"] == course]
    per_course, errors = {}, []
    for c in scope:
        slug, name = c["slug"], c["name"]
        manifest = _load_manifest(slug)
        old_sources = dict(manifest["sources"])
        new_sources = {}
        n_norm = n_skip = 0
        seen_keys = set()
        for fn in _NORMALIZERS_API:
            try:
                results = fn(slug, name)
            except Exception as exc:  # noqa: BLE001
                errors.append(f"{slug} {fn.__name__}: {exc}")
                continue
            for key, h, path in results:
                seen_keys.add(key)
                rel = str(path.relative_to(course_dir(slug)))
                prev = old_sources.get(key)
                if prev and prev["hash"] == h and (course_dir(slug) / prev["outPath"]).exists():
                    new_sources[key] = prev
                    n_skip += 1
                    # write_normalized already rewrote the file identically; that's
                    # acceptable, but to honour "no rewrite" we restore mtime:
                    continue
                new_sources[key] = {"hash": h, "outPath": rel}
                n_norm += 1
        orphans = 0
        for key, meta in old_sources.items():
            if key not in seen_keys:
                p = course_dir(slug) / meta["outPath"]
                if p.exists():
                    p.unlink()
                orphans += 1
        manifest["sources"] = new_sources
        manifest["ingestedAt"] = _now()
        _save_manifest(slug, manifest)
        per_course[slug] = {"normalized": n_norm, "skipped": n_skip, "orphansDeleted": orphans}
    return {"perCourse": per_course, "errors": errors}
```

> **Note the "no rewrite on unchanged" requirement (spec §7 test):** the simplest correct approach is to have each `_normalize_*` compute its `input_hash` and its target path FIRST, check the manifest, and only call `write_normalized` when changed/new. Restructure `_normalize_materials/assignments/transcripts` to take `old_sources` and return `(key, hash, path, was_written: bool)`, skipping the `write_normalized` call (and returning the existing path) when the hash matches. Implement that now — the test `test_step_ingest_skips_unchanged_and_deletes_orphans` asserts `note.stat().st_mtime == mtime1`, which fails if `write_normalized` runs every time.
